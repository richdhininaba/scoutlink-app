      var position = window.ScoutLinkScoringV4.normalisePosition(el('cb4Position').value);
      if (!firstName || !lastName || !ageGroup || !position) throw new Error('First name, last name, age group and exact position are required.');
      row.firstName = firstName;
      row.lastName = lastName;
      row.ageGroup = ageGroup;
      row.position = position;
      row.foot = el('cb4Foot').value;
      row.heightCategory = el('cb4Height').value;
      row.buildCategory = el('cb4Build').value;
      row.assignedCoachId = coaches.length ? el('cb4Coach').value : defaultCoachId;
      row.attributeRatings = modalRatings();
      closeEdit();
      render();
    } catch (error) {
      el('cb4ModalError').textContent = error.message;
      el('cb4ModalError').classList.add('is-visible');
    }
  }

  function clearMessages() {
    if (el('formError')) { el('formError').textContent = ''; el('formError').className = 'ap3-message is-error'; }
    if (el('formSuccess')) { el('formSuccess').textContent = ''; el('formSuccess').className = 'ap3-message is-success'; }
  }
  function showError(message) {
    clearMessages(); el('formError').textContent = message; el('formError').className = 'ap3-message is-error show';
  }
  function showSuccess(message) {
    clearMessages(); el('formSuccess').textContent = message; el('formSuccess').className = 'ap3-message is-success show';
  }

  function payloadFor(row) {
    return {
      firstName:row.firstName.trim(), lastName:row.lastName.trim(), ageGroup:row.ageGroup,
      positionGroup:window.ScoutLinkScoringV4.groupForPosition(row.position),
      specificPosition:row.position, primaryPosition:row.position, positions:[row.position],
      foot:row.foot || 'Right', heightCategory:row.heightCategory || 'average',
      buildCategory:row.buildCategory || 'athletic', assignedCoachId:row.assignedCoachId || null,
      attributeRatings:row.attributeRatings || {}, attribute_rating_scale:'ten'
    };
  }
