    );
  }


  function normaliseFoot(value) {
    var foot = String(value || '').trim().toLowerCase();
    if (foot === 'left') return 'Left';
    if (foot === 'both' || foot === 'two-footed' || foot === 'two footed') return 'Both';
    return 'Right';
  }

  function normaliseProfile(value, allowed, fallback) {
    var candidate = String(value || '').trim().toLowerCase().replace(/[^a-z0-9]+/g, '_');
    return allowed.some(function (option) { return option[0] === candidate; }) ? candidate : fallback;
  }

  function blankRow() {
    return {
      id: 'bulk-v4-' + nextId++,
      firstName: '',
      lastName: '',
      ageGroup: '',
      position: '',
      foot: 'Right',
      heightCategory: 'average',
      buildCategory: 'athletic',
      assignedCoachId: defaultCoachId,
      attributeRatings: {},
      attributesExpanded: true,
      sourceRow: null
    };
  }

  function playerName(row) {
    return [row.firstName, row.lastName].filter(Boolean).join(' ') || 'Unnamed player';
  }

  function flatRatings(row) {
    return window.ScoutLinkScoringV4.flattenRatings(row.attributeRatings || {});
  }

  function expectedAttributes(row) {
    return row.position
      ? window.ScoutLinkScoringV4.attributesForPosition(row.position, options)
      : [];
  }

  function coreErrors(row) {
    var errors = [];
    if (!String(row.firstName || '').trim()) errors.push('first name');
    if (!String(row.lastName || '').trim()) errors.push('last name');
    if (!options.ageGroups.includes(row.ageGroup)) errors.push('age group');
    if (!window.ScoutLinkScoringV4.normalisePosition(row.position)) errors.push('supported position');
    return errors;
  }

  function status(row) {
    var errors = coreErrors(row);
    var expected = expectedAttributes(row).length;
    var observed = Object.keys(flatRatings(row)).filter(function (key) {
      return expectedAttributes(row).some(function (attribute) {
        return attribute[0] === key;
      });
    }).length;

    return {
      ready: errors.length === 0,
      text: errors.length
        ? 'Add ' + errors.join(', ')
        : observed + ' of ' + expected + ' attributes observed',
      observed: observed,
      expected: expected
    };
  }

  function ratingLabel(option) {
    return String(option.label || '').replace(/^\s*\d+\s*[—–-]\s*/, '');
  }

  function optionMarkup(values, selected, placeholder) {
    var html = placeholder == null
      ? ''
      : '<option value="">' + esc(placeholder) + '</option>';

    html += values.map(function (item) {
      var value = Array.isArray(item) ? item[0] : item.value;
      var label = Array.isArray(item) ? item[1] : item.label;
      if (typeof item === 'string') {
        value = item;
        label = item;
      }
      return '<option value="' + esc(value) + '"' +
        (String(value) === String(selected == null ? '' : selected) ? ' selected' : '') +
        '>' + esc(label) + '</option>';
    }).join('');

    return html;
  }
