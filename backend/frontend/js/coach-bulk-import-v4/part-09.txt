    link.click();
    link.remove();
    URL.revokeObjectURL(url);

  function normaliseHeader(value) {
    return String(value || '')
      .trim()
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, '_')
      .replace(/^_|_$/g, '');
  }

  function validateImportHeaders(record) {
    var headers = Object.keys(record || {}).map(normaliseHeader);
    var missing = REQUIRED_IMPORT_HEADERS.filter(function (header) {
      return !headers.includes(header);
    });
    var unsupported = headers.filter(function (header) {
      return header && !TEMPLATE_HEADERS.includes(header);
    });

    if (missing.length) {
      throw new Error('The file is missing these required columns: ' + missing.join(', ') + '.');
    }
    if (unsupported.length) {
      throw new Error(
        'The player-details file may contain only: ' + TEMPLATE_HEADERS.join(', ') +
        '. Remove these unsupported columns: ' + unsupported.join(', ') + '.'
      );
    }
  }

  function firstValue(record, keys) {
    for (var index = 0; index < keys.length; index += 1) {
      var value = record[keys[index]];
      if (value !== undefined && value !== null && String(value).trim() !== '') return value;
    }
    return '';
  }

  function rowFromRecord(record, rowIndex) {
    var normalised = {};
    Object.keys(record || {}).forEach(function (key) {
      normalised[normaliseHeader(key)] = record[key];
    });

    var row = blankRow();
    row.sourceRow = rowIndex + 2;
    row.firstName = String(firstValue(normalised, ['first_name'])).trim();
    row.lastName = String(firstValue(normalised, ['last_name'])).trim();
    row.ageGroup = String(firstValue(normalised, ['age_group'])).toUpperCase();
    row.position = window.ScoutLinkScoringV4.normalisePosition(firstValue(normalised, ['position'])) || '';
    row.foot = normaliseFoot(firstValue(normalised, ['foot']));
    row.heightCategory = normaliseProfile(firstValue(normalised, ['height_category']), HEIGHT_OPTIONS, 'average');
    row.buildCategory = normaliseProfile(firstValue(normalised, ['build_category']), BUILD_OPTIONS, 'athletic');
    row.attributeRatings = {};
    row.attributesExpanded = true;
    return row;
  }

  function parseCsv(text) {
    var lines = String(text || '')
      .replace(/^\uFEFF/, '')
      .split(/\r?\n/)
      .filter(function (line) { return line.trim(); });
    if (!lines.length) return [];

    function parseLine(line) {
      var values = [];
      var current = '';
      var quoted = false;
      for (var index = 0; index < line.length; index += 1) {
        var character = line[index];
        if (character === '"' && quoted && line[index + 1] === '"') {
          current += '"';
          index += 1;
        } else if (character === '"') {
          quoted = !quoted;
        } else if (character === ',' && !quoted) {
          values.push(current);
          current = '';
        } else {
          current += character;
        }
      }
      values.push(current);
      return values;
    }

    var headers = parseLine(lines[0]);
    return lines.slice(1).map(function (line) {
      var values = parseLine(line);
      var record = {};
      headers.forEach(function (header, index) {
        record[header] = values[index] == null ? '' : values[index];
      });
      return record;
    });
