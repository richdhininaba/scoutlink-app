  function buildPage() {
    document.body.classList.add('coach-bulk-import-v3');
    var page = document.querySelector('.page-content');
    if (!page) return;
    var banner = page.querySelector('.public-demo-banner');
    page.innerHTML = shellMarkup();
    if (banner) page.insertBefore(banner, page.firstChild);
    installStyles();

    var title = document.querySelector('.topbar-title');
    if (title) title.textContent = 'Bulk import';
    var mobileTitle = document.querySelector('.coach-v2-mobile-title');
    if (mobileTitle) mobileTitle.textContent = 'Bulk import';

    bindActions();
    render();
    loadCoachContext();
  }

  function teamName() {
    return el('teamNameLabel') ? el('teamNameLabel').textContent : 'My team';
  }

  function coachName(id) {
    var coach = coaches.find(function (item) {
      return String(item.id) === String(id);
    });
    return coach
      ? [coach.first_name, coach.last_name].filter(Boolean).join(' ')
      : 'Current Coach';
  }
