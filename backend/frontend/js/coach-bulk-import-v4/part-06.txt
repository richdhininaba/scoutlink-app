  function render() {
    var ready = rows.filter(function (row) { return status(row).ready; }).length;
    var review = rows.length - ready;
    [['cb3TotalRows',rows.length],['cb3ReadyRows',ready],['cb3ReviewRows',review],['cb3ReadyPill',ready],['cb3ReviewPill',review]].forEach(function (pair) {
      if (el(pair[0])) el(pair[0]).textContent = pair[1];
    });
    if (el('cb3FileName')) el('cb3FileName').textContent = fileName ? 'Selected file: ' + fileName : '';
    el('cb3PreviewBody').innerHTML = rows.map(previewRow).join('');
    el('cb3PreviewEmpty').style.display = rows.length ? 'none' : 'grid';
    var issues = rows.filter(function (row) { return !status(row).ready; });
    if (issues.length) {
      el('cb3IssueBody').innerHTML = '<div class="cb3-issue-list">' + issues.map(function (row) {
        return '<article class="cb3-issue"><div><b>' + esc(playerName(row)) + '</b><p>' + esc(status(row).text) +
          '</p></div><button class="cb3-btn is-primary is-small" type="button" data-edit-row="' + esc(row.id) + '">Fix row</button></article>';
      }).join('') + '</div>';
    } else if (rows.length) {
      el('cb3IssueBody').innerHTML = '<div class="cb3-all-ready"><span class="cb3-pill is-green">Ready</span>' +
        '<div><strong>Every row passes the current validation.</strong><p>Review the compact preview once more, then submit all players.</p></div></div>';
    } else {
      el('cb3IssueBody').innerHTML = '<div class="cb3-preview-empty">No rows have been added yet.</div>';
    }
    bindDynamicActions();
  }
