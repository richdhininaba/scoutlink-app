  }

  async function importFile(file) {
    clearMessages();
    fileName = file.name;
    try {
      var records;
      if (/\.csv$/i.test(file.name)) {
        records = parseCsv(await file.text());
      } else {
        if (!window.XLSX) {
          throw new Error('Spreadsheet support did not load. Save the file as CSV or refresh and try again.');
        }
        var buffer = await file.arrayBuffer();
        var workbook = window.XLSX.read(buffer, { type: 'array' });
        records = window.XLSX.utils.sheet_to_json(
          workbook.Sheets[workbook.SheetNames[0]],
          { defval: '' }
        );
      }

      if (!records.length) throw new Error('The file does not contain player rows.');
      validateImportHeaders(records[0]);
      rows = records.slice(0, MAX_ROWS).map(rowFromRecord);
      render();
      showSuccess(
        rows.length + ' player row' + (rows.length === 1 ? '' : 's') +
        ' imported. Complete the position-specific attributes directly in the table.'
      );
    } catch (error) {
      showError(error.message || 'The file could not be imported.');
    }
  }

  async function loadCoachContext() {
    if (isPublicDemo()) {
      el('teamNameLabel').textContent = 'Public demo session';
      return;
    }
    if (!window.Auth || window.Auth.type !== 'Coach') {
      el('teamNameLabel').textContent = 'Stratex managed import';
      return;
    }

    try {
      var profile = await window.api('GET', '/api/coaches/profile');
      var coach = profile.coach || {};
      el('teamNameLabel').textContent = coach.team_name || 'My team';
      defaultCoachId = coach.id || '';

      if (coach.is_super_user) {
        var response = await window.api('GET', '/api/coaches/team-coaches');
        coaches = [coach].concat(response.coaches || response.data || []).filter(function (row, index, all) {
          return row && row.id && all.findIndex(function (other) {
            return other && other.id === row.id;
          }) === index;
        });
        el('defaultCoachSelect').innerHTML = coachOptionsMarkup(defaultCoachId);
        el('defaultCoachField').style.display = '';
      }
    } catch (error) {
      el('teamNameLabel').textContent = 'Team unavailable';
      showError('Team assignment could not be loaded: ' + error.message);
    }
    render();
  }

  async function init() {
    if (!allowedContext()) {
      window.location.href = '/login';
      return;
    }
    if (!window.ScoutLinkScoringV4) {
      window.setTimeout(init, 40);
      return;
    }

    options = await window.ScoutLinkScoringV4.loadOptions();
    if (typeof window.buildScoutNav === 'function') {
      window.buildScoutNav('sidebarNav', 'Coach');
    }
    buildPage();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init, { once: true });
  } else {
    init();
  }
