        response = await window.api('POST','/api/players/bulk',{players:rows.map(payloadFor)});
      }
      var body = response.data || response;
      var createdCount = Array.isArray(body.created) ? body.created.length : rows.length;
      var errorCount = Array.isArray(body.errors) ? body.errors.length : 0;
      el('resultsBox').innerHTML = '<div class="cb3-all-ready"><span class="cb3-pill is-green">Complete</span><div>' +
        '<strong>' + createdCount + ' player' + (createdCount === 1 ? '' : 's') + ' added.</strong>' +
        '<p>' + (errorCount ? errorCount + ' row(s) returned an error.' : 'Every submitted player now uses the V4 position-aware scoring structure.') + '</p></div></div>';
      showSuccess(createdCount + ' player' + (createdCount === 1 ? '' : 's') + ' added successfully.');
      rows = [];
      render();
    } catch (error) {
      showError(error.message || 'The players could not be submitted.');
    } finally {
      buttons.forEach(function (button) { button.disabled = false; button.textContent = 'Submit all players'; });
    }
  }

  function allAttributeKeys() {
    return unique(['general','goalkeeper','defender','midfielder','attacker'].flatMap(function (group) {
      return (options.attributes[group] || []).map(function (row) { return row[0]; });
    }));
  }

  function downloadTemplate() {
    var headers = ['first_name','last_name','age_group','position','foot','height_category','build_category'].concat(allAttributeKeys());
    var csv = headers.join(',') + '\n';
    var blob = new Blob([csv], {type:'text/csv;charset=utf-8'});
    var url = URL.createObjectURL(blob);
    var link = document.createElement('a');
    link.href = url;
    link.download = 'scoutlink-v4-player-import-template.csv';
    document.body.appendChild(link);
    link.click();
    link.remove();
    URL.revokeObjectURL(url);
  }

  function normaliseHeader(value) {
    return String(value || '').trim().toLowerCase().replace(/[^a-z0-9]+/g,'_').replace(/^_|_$/g,'');
  }
  function firstValue(record, keys) {
    for (var index = 0; index < keys.length; index += 1) {
      var value = record[keys[index]];
      if (value !== undefined && value !== null && String(value).trim() !== '') return value;
    }
    return '';
  }
