  function positionOptionsMarkup(selected) {
    return '<option value="">Select position</option>' + options.positionGroups.map(function (group) {
      var positions = options.positions.filter(function (position) {
        return position.group === group.key;
      });
      return '<optgroup label="' + esc(group.label) + '">' + positions.map(function (position) {
        return '<option value="' + esc(position.code) + '"' +
          (position.code === selected ? ' selected' : '') + '>' +
          esc(position.label + ' (' + position.code + ')') + '</option>';
      }).join('') + '</optgroup>';
    }).join('');
  }

  function coachOptionsMarkup(selected) {
    return coaches.map(function (coach) {
      var name = [coach.first_name, coach.last_name].filter(Boolean).join(' ');
      return '<option value="' + esc(coach.id) + '"' +
        (String(coach.id) === String(selected || defaultCoachId) ? ' selected' : '') +
        '>' + esc(name) + '</option>';
    }).join('');
  }
