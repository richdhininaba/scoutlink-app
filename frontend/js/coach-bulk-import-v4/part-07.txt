      if (!ratingRow || !ratingRow.position) return;
      try {
        var flat = flatRatings(ratingRow);
        var rating = window.ScoutLinkScoringV4.validateRating(ratingField.value);
        if (rating === null) delete flat[ratingField.dataset.ratingKey];
        else flat[ratingField.dataset.ratingKey] = rating;
        ratingRow.attributeRatings = window.ScoutLinkScoringV4.nestRatings(
          flat,
          window.ScoutLinkScoringV4.groupForPosition(ratingRow.position),
          options
        );
        refreshRowStatus(ratingRow);
      } catch (error) {
        showError(error.message);
      }
    });
  }

  function bindActions() {
    document.querySelectorAll('[data-bulk-add]').forEach(function (button) {
      button.addEventListener('click', function () {
        if (rows.length >= MAX_ROWS) {
          showError('A maximum of 50 players can be imported at once.');
          return;
        }
        var row = blankRow();
        rows.push(row);
        render();
        window.setTimeout(function () {
          var input = document.querySelector('[data-core-row="' + row.id + '"] [data-row-field="firstName"]');
          if (input) {
            input.scrollIntoView({ behavior: 'smooth', block: 'center' });
            input.focus();
          }
        }, 0);
      });
    });

    document.querySelectorAll('[data-bulk-choose]').forEach(function (button) {
      button.addEventListener('click', function () {
        el('cb4FileInput').click();
      });
    });

    document.querySelectorAll('[data-bulk-download]').forEach(function (button) {
      button.addEventListener('click', downloadTemplate);
    });

    document.querySelectorAll('[data-bulk-submit]').forEach(function (button) {
      button.addEventListener('click', submitAll);
    });

    el('cb4FileInput').addEventListener('change', function () {
      var file = el('cb4FileInput').files && el('cb4FileInput').files[0];
      if (file) importFile(file);
      el('cb4FileInput').value = '';
    });

    var drop = el('cb3DropZone');
    ['dragenter', 'dragover'].forEach(function (name) {
      drop.addEventListener(name, function (event) {
        event.preventDefault();
        drop.classList.add('is-dragging');
      });
    });
    ['dragleave', 'drop'].forEach(function (name) {
      drop.addEventListener(name, function (event) {
        event.preventDefault();
        drop.classList.remove('is-dragging');
      });
    });
    drop.addEventListener('drop', function (event) {
      var file = event.dataTransfer && event.dataTransfer.files && event.dataTransfer.files[0];
      if (file) importFile(file);
    });
    drop.addEventListener('keydown', function (event) {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        el('cb4FileInput').click();
      }
    });

    el('defaultCoachSelect').addEventListener('change', function () {
      defaultCoachId = el('defaultCoachSelect').value;
      rows.forEach(function (row) {
        if (!row.assignedCoachId) row.assignedCoachId = defaultCoachId;
      });
      render();
    });

    bindTableEditing();
  }
